```typescript
import React from "react";
import { interpolate, useCurrentFrame, Easing } from "remotion";
import { HEKAYA_PALETTE } from "../schema";

interface Props {
  text: string;
  startFrame: number;
  durationFrames: number;
  /** Optional word to flash gold inside the phrase. */
  keyWord?: string;
  /** Pixel font size. Default 56. */
  fontSize?: number;
  /** Max width in pixels. Default 920. */
  maxWidth?: number;
  /** Right alignment for RTL. Default true. */
  rtl?: boolean;
}

/**
 * Phrase reveal via clip-path mask wipe (right-to-left for Arabic).
 * Reveals 0→100% over the first 14 frames, holds, then fades out
 * over the last 10 frames. The keyWord flashes gold for 4 frames
 * at completion of the reveal.
 */
export const PhraseReveal: React.FC<Props> = ({
  text,
  startFrame,
  durationFrames,
  keyWord,
  fontSize = 56,
  maxWidth = 920,
  rtl = true,
}) => {
  const frame = useCurrentFrame();
  const localFrame = frame - startFrame;

  // Reveal: clip-path inset wipes from right (100%) to left (0%) over 14f
  const reveal = interpolate(localFrame, [0, 14], [100, 0], {
    extrapolateLeft: "clamp",
    extrapolateRight: "clamp",
    easing: Easing.bezier(0.25, 0, 0.25, 1),
  });

  // Fade out at the end of the phrase duration
  const fadeOut = interpolate(
    localFrame,
    [durationFrames - 10, durationFrames],
    [1, 0],
    { extrapolateLeft: "clamp", extrapolateRight: "clamp" },
  );

  // Don't render before start or after end
  if (localFrame < 0 || localFrame > durationFrames) return null;

  // Highlight the keyWord with gold for frames 14-18 (just after reveal lands)
  const highlightActive = localFrame >= 14 && localFrame <= 18;
  const renderText = () => {
    if (!keyWord || !text.includes(keyWord)) {
      return text;
    }
    const parts = text.split(keyWord);
    return (
      <>
        {parts.map((part, i) => (
          <React.Fragment key={i}>
            {part}
            {i < parts.length - 1 && (
              <span
                style={{
                  color: highlightActive ? HEKAYA_PALETTE.gold : HEKAYA_PALETTE.ivory,
                  transition: "color 80ms ease-out",
                }}
              >
                {keyWord}
              </span>
            )}
          </React.Fragment>
        ))}
      </>
    );
  };

  return (
    <div
      dir={rtl ? "rtl" : "ltr"}
      style={{
        fontFamily: "Tajawal, system-ui, sans-serif",
        fontSize,
        fontWeight: 700,
        color: HEKAYA_PALETTE.ivory,
        maxWidth,
        lineHeight: 1.4,
        textAlign: rtl ? "right" : "left",
        // Mask wipe — for RTL, inset from right edge
        clipPath: rtl
          ? `inset(0 ${reveal}% 0 0)`
          : `inset(0 0 0 ${reveal}%)`,
        opacity: fadeOut,
        textShadow: `0 4px 18px ${HEKAYA_PALETTE.shadow}cc`,
      }}
    >
      {renderText()}
    </div>
  );
};
```

### Task 3: ParallaxImage component (2.5D depth on stills)

**Files:**
- Create: `my-video/src/compositions/Hekaya2/components/ParallaxImage.tsx`

- [ ] **Step 1: Create the component**

```typescript
import React from "react";
import { AbsoluteFill, Img, interpolate, useCurrentFrame, Easing, staticFile } from "remotion";
import { HEKAYA_PALETTE } from "../schema";

interface Props {
  src: string;
  durationFrames: number;
  /** Direction of the slow camera move. Default "in". */
  direction?: "in" | "out" | "left" | "right";
  /** Apply duotone treatment (gold + dusk-blue). Default false. */
  duotone?: boolean;
  /** Apply dark vignette + warm overlay. Default true. */
  vignette?: boolean;
}

/**
 * Layered photo treatment: background (full image) translates / scales slowly
 * for a Ken-Burns-style atmospheric drift; foreground holds a higher-contrast
 * version slightly offset to create a 2.5D parallax illusion. Adds optional
 * duotone for archival treatment, and a warm vignette overlay.
 */
export const ParallaxImage: React.FC<Props> = ({
  src,
  durationFrames,
  direction = "in",
  duotone = false,
  vignette = true,
}) => {
  const frame = useCurrentFrame();
  const progress = interpolate(frame, [0, durationFrames], [0, 1], {
    extrapolateRight: "clamp",
  });

  // Background drift
  let bgScale = 1;
  let bgTx = 0;
  let bgTy = 0;

  if (direction === "in") {
    bgScale = interpolate(progress, [0, 1], [1.0, 1.10], {
      easing: Easing.bezier(0.25, 0, 0.25, 1),
    });
  } else if (direction === "out") {
    bgScale = interpolate(progress, [0, 1], [1.16, 1.04]);
  } else if (direction === "left") {
    bgScale = 1.12;
    bgTx = interpolate(progress, [0, 1], [-2.5, 2.5]);
  } else {
    bgScale = 1.12;
    bgTx = interpolate(progress, [0, 1], [2.5, -2.5]);
  }

  // Foreground moves opposite for parallax
  const fgScale = bgScale * 1.04;
  const fgTx = -bgTx * 0.4;
  const fgTy = -bgTy * 0.4;

  const resolvedSrc = src.startsWith("http") ? src : staticFile(src);

  const filter = duotone
    ? "grayscale(60%) sepia(20%) hue-rotate(-30deg) brightness(0.92)"
    : undefined;

  return (
    <AbsoluteFill style={{ overflow: "hidden", backgroundColor: HEKAYA_PALETTE.shadow }}>
      {/* Background layer — slower drift */}
      <Img
        src={resolvedSrc}
        style={{
          width: "100%",
          height: "100%",
          objectFit: "cover",
          transform: `scale(${bgScale}) translate(${bgTx}%, ${bgTy}%)`,
          filter,
        }}
      />
      {/* Foreground layer — opposite drift, slightly cropped tighter */}
      <AbsoluteFill style={{ opacity: 0.45 }}>
        <Img
          src={resolvedSrc}
          style={{
            width: "100%",
            height: "100%",
            objectFit: "cover",
            transform: `scale(${fgScale}) translate(${fgTx}%, ${fgTy}%)`,
            filter,
          }}
        />
      </AbsoluteFill>

      {vignette ? (
        <>
          <AbsoluteFill
            style={{
              background: `radial-gradient(ellipse at center,
                transparent 35%,
                ${HEKAYA_PALETTE.ink}99 75%,
                ${HEKAYA_PALETTE.ink} 100%)`,
            }}
          />
          <AbsoluteFill
            style={{
              background: `linear-gradient(180deg,
                ${HEKAYA_PALETTE.ink}40 0%,
                transparent 38%,
                transparent 62%,
                ${HEKAYA_PALETTE.ink}cc 100%)`,
            }}
          />
        </>
      ) : null}
    </AbsoluteFill>
  );
};
```

```typescript
import React from "react";
import { AbsoluteFill, interpolate, useCurrentFrame } from "remotion";
import { HEKAYA_PALETTE } from "../schema";

/**
 * Atmospheric tier 1 — gold particles drifting upward like dust in a sunbeam.
 * Decoupled from v1 so v2 can evolve the look. Currently 22 particles with
 * randomized seeds, drifting up over 600 frames with fade in/out.
 */
export const Embers: React.FC<{ count?: number }> = ({ count = 22 }) => {
  const frame = useCurrentFrame();
  const dots = Array.from({ length: count }, (_, i) => i);
  return (
    <AbsoluteFill style={{ pointerEvents: "none" }}>
      {dots.map((i) => {
        const seed = (i * 37) % 100;
        const x = (seed * 13) % 1080;
        const y = ((seed * 47) % 1920);
        const driftY = interpolate(frame, [0, 600], [0, -120 - (seed % 40)]);
        const opacity = interpolate(
          frame,
          [0, 60, 540, 600],
          [0, 0.45 + (seed % 30) / 200, 0.45 + (seed % 30) / 200, 0],
          { extrapolateRight: "extend" },
        );
        return (
          <div
            key={i}
            style={{
              position: "absolute",
              left: x,
              top: y + driftY,
              width: 3 + (seed % 3),
              height: 3 + (seed % 3),
              borderRadius: "50%",
              background: HEKAYA_PALETTE.gold,
              boxShadow: `0 0 8px ${HEKAYA_PALETTE.gold}`,
              opacity,
            }}
          />
        );
      })}
    </AbsoluteFill>
  );
};
```

```typescript
import React from "react";
import { AbsoluteFill, useCurrentFrame } from "remotion";
import { ChapterCycleProps, PhraseMarkerProps, HEKAYA_PALETTE } from "../schema";
import { ParallaxImage } from "../components/ParallaxImage";
import { Embers } from "../components/Embers";
import { PhraseReveal } from "../components/PhraseReveal";

interface Props {
  cycle: ChapterCycleProps;
  /** All phrase markers — this scene picks the ones inside its frame range. */
  phrases: PhraseMarkerProps[];
  /** Absolute frame offset from composition start. */
  absoluteStart: number;
}

/**
 * The workhorse scene. Cycles through 5-7 photos with 2.5s cuts, layered
 * with phrase reveals timed to the VO. Each photo gets its own KB direction.
 */
export const Chapter: React.FC<Props> = ({ cycle, phrases, absoluteStart }) => {
  const frame = useCurrentFrame();
  const localFrame = frame; // we're inside a Sequence, frame is relative

  // Compute which photo to show based on cycle progress.
  const cuts = cycle.photos.length;
  const perPhotoFrames = Math.floor(cycle.durationFrames / cuts);
  const currentPhotoIdx = Math.min(Math.floor(localFrame / perPhotoFrames), cuts - 1);
  const photo = cycle.photos[currentPhotoIdx];

  // Direction rotates per photo so cumulative motion is varied
  const directions: Array<"in" | "out" | "left" | "right"> = ["in", "out", "left", "right"];
  const direction = directions[currentPhotoIdx % directions.length];

  // Filter phrases relevant to this scene window
  const sceneEnd = absoluteStart + cycle.durationFrames;
  const myPhrases = phrases.filter(
    (p) =>
      p.startFrame >= absoluteStart &&
      p.startFrame < sceneEnd
  );

  return (
    <AbsoluteFill style={{ backgroundColor: HEKAYA_PALETTE.ink }}>
      <ParallaxImage
        src={photo}
        durationFrames={perPhotoFrames}
        direction={direction}
        // Archival treatment on the texture frames (every 3rd in the cycle)
        duotone={currentPhotoIdx % 3 === 2}
      />
      <Embers />

      {myPhrases.map((p, i) => {
        const localStart = p.startFrame - absoluteStart;
        return (
          <div
            key={i}
            style={{
              position: "absolute",
              bottom: p.layoutType === "lower-third" ? 320 : "auto",
              top: p.layoutType === "full-bleed-quote" ? "32%" : "auto",
              left: 70,
              right: 70,
              textAlign: p.layoutType === "full-bleed-quote" ? "center" : "right",
            }}
          >
            <PhraseReveal
              text={p.arabicText}
              startFrame={localStart}
              durationFrames={p.durationFrames}
              keyWord={p.keyWord}
              fontSize={
                p.layoutType === "single-word"
                  ? 140
                  : p.layoutType === "full-bleed-quote"
                  ? 72
                  : 52
              }
            />
          </div>
        );
      })}
    </AbsoluteFill>
  );
};
```

```typescript
import React from "react";
import { AbsoluteFill, interpolate, useCurrentFrame, Easing } from "remotion";
import { Hekaya2Props, HEKAYA_PALETTE } from "../schema";
import { ParallaxImage } from "../components/ParallaxImage";
import { Embers } from "../components/Embers";

interface Props {
  closingMedia: string;
  loopHook: Hekaya2Props["loopHook"];
  handle: string;
  sources: Hekaya2Props["sources"];
  durationFrames: number;
}

/**
 * Final 10s — closing image + loop callback. The closing phrase poses a
 * question that the opening phrase answered, so the loop feels causal,
 * not repetitive.
 */
export const Resonance: React.FC<Props> = ({
  closingMedia,
  loopHook,
  handle,
  sources,
  durationFrames,
}) => {
  const frame = useCurrentFrame();

  const closingOpacity = interpolate(frame, [0, 30], [0, 1], {
    extrapolateRight: "clamp",
  });
  const closingY = interpolate(frame, [0, 40], [20, 0], {
    easing: Easing.bezier(0.16, 1, 0.3, 1),
    extrapolateRight: "clamp",
  });

  // Handle + sources fade in toward the end
  const handleOpacity = interpolate(
    frame,
    [80, 120, durationFrames - 6, durationFrames],
    [0, 1, 1, 1],  // sustain to the cold-cut, no fade
    { extrapolateRight: "clamp" },
  );

  return (
    <AbsoluteFill style={{ backgroundColor: HEKAYA_PALETTE.ink }}>
      <ParallaxImage src={closingMedia} durationFrames={durationFrames} direction="out" />
      <Embers />

      {/* Closing phrase — the loop hook */}
      <div
        style={{
          position: "absolute",
          top: "32%",
          left: 80,
          right: 80,
          textAlign: "center",
          opacity: closingOpacity,
          transform: `translateY(${closingY}px)`,
        }}
      >
        <div
          dir="rtl"
          style={{
            color: HEKAYA_PALETTE.ivory,
            fontFamily: "Tajawal, system-ui, sans-serif",
            fontSize: 72,
            fontWeight: 800,
            lineHeight: 1.4,
            textShadow: `0 4px 20px ${HEKAYA_PALETTE.shadow}`,
          }}
        >
          {loopHook.closingPhrase}
        </div>
      </div>

      {/* Handle + sources strip — sustains until cold cut */}
      <div
        style={{
          position: "absolute",
          bottom: 220,
          left: 0,
          right: 0,
          textAlign: "center",
          opacity: handleOpacity,
        }}
      >
        <div
          style={{
            color: HEKAYA_PALETTE.ivory,
            fontFamily: "system-ui, -apple-system, sans-serif",
            fontSize: 28,
            letterSpacing: 4,
            fontWeight: 500,
            opacity: 0.92,
          }}
        >
          {handle}
        </div>
        <div
          style={{
            color: HEKAYA_PALETTE.gold,
            fontFamily: "system-ui, -apple-system, sans-serif",
            fontSize: 13,
            letterSpacing: 3,
            marginTop: 14,
            opacity: 0.6,
            textTransform: "uppercase",
          }}
        >
          {sources
            .slice(0, 4)
            .map((s) => s.name)
            .join(" · ")}
        </div>
      </div>
    </AbsoluteFill>
  );
};
```

```typescript
import React from "react";
import {
  AbsoluteFill,
  Audio,
  Sequence,
  interpolate,
  useCurrentFrame,
  staticFile,
} from "remotion";
import {
  Hekaya2Props,
  HEKAYA2_TOTAL_FRAMES,
  HEKAYA2_PHASES,
  HEKAYA_PALETTE,
} from "./schema";
import { ColdOpen } from "./scenes/ColdOpen";
import { Chapter } from "./scenes/Chapter";
import { SilencePivot } from "./scenes/SilencePivot";
import { Resonance } from "./scenes/Resonance";

/**
 * HEKAYA v2 — narrated mini-documentary, 75 seconds.
 *
 * Audio architecture: VO (-6 dB) + Music ducked (-18 dB) / swell (-10 dB) +
 * SFX/foley markers (-22 dB). Full silence at SILENCE_PIVOT phase (1s).
 *
 * Visual architecture: ColdOpen (0-8s) → 3 Chapter scenes with photo cycles
 * (8-45s) → SilencePivot (45-46s) → Chapter for climax+resolution (46-65s) →
 * Resonance (65-75s, cold-cut at 1:15).
 */
export const Hekaya2: React.FC<Hekaya2Props> = ({
  dateLabel,
  arabicDateLabel,
  handle,
  voiceOver,
  music,
  sfx,
  title,
  phrases,
  heroMedia,
  chapters,
  closingMedia,
  loopHook,
  sources,
}) => {
  const frame = useCurrentFrame();

  // Music volume: ducked under VO at -18 dB equivalent (~0.13 linear),
  // swells -10 dB (~0.32) at chapter transitions, full silence during pivot.
  const musicVol = (() => {
    // Full silence during the pivot phase
    if (
      frame >= HEKAYA2_PHASES.SILENCE_PIVOT.start &&
      frame < HEKAYA2_PHASES.SILENCE_PIVOT.end
    ) {
      return 0;
    }
    // Cold-open: music enters at frame 36 (1.2s)
    if (frame < 36) return 0;
    if (frame < 48) {
      return interpolate(frame, [36, 48], [0, 0.13], {
        extrapolateRight: "clamp",
      });
    }
    // Climax swells music to -10 dB
    if (
      frame >= HEKAYA2_PHASES.CLIMAX.start &&
      frame < HEKAYA2_PHASES.CLIMAX.start + 60
    ) {
      return interpolate(
        frame,
        [HEKAYA2_PHASES.CLIMAX.start, HEKAYA2_PHASES.CLIMAX.start + 60],
        [0.13, 0.32],
        { extrapolateRight: "clamp" },
      );
    }
    if (frame >= HEKAYA2_PHASES.CLIMAX.start + 60 && frame < HEKAYA2_PHASES.RESOLUTION.start) {
      return 0.32;
    }
    // Cold-cut at end (no fade)
    if (frame >= HEKAYA2_TOTAL_FRAMES - 5) return 0;
    return 0.13;
  })();

  // VO volume: full at -6 dB equivalent (~0.5 linear), silent during pivot
  const voVol = (() => {
    if (
      frame >= HEKAYA2_PHASES.SILENCE_PIVOT.start &&
      frame < HEKAYA2_PHASES.SILENCE_PIVOT.end
    ) {
      return 0;
    }
    if (frame >= HEKAYA2_TOTAL_FRAMES - 5) return 0;
    return 0.5;
  })();

  const voSrc = voiceOver.startsWith("http") ? voiceOver : staticFile(voiceOver);
  const musicSrc = music.startsWith("http") ? music : staticFile(music);

  // Chapter timing — divide the 8-45s window into 3 chapters by their durations
  const chapter1Start = HEKAYA2_PHASES.WANT.start;          // 240
  const chapter2Start = HEKAYA2_PHASES.OBSTACLE_1.start;    // 525  (this is wrong — need to re-think)
  // Actually: chapters[0] covers WANT+OBSTACLE_1, chapters[1] covers ESCALATION_1, chapters[2] covers ESCALATION_2

  return (
    <AbsoluteFill style={{ backgroundColor: HEKAYA_PALETTE.ink }}>
      <Audio src={voSrc} volume={voVol} />
      <Audio src={musicSrc} volume={musicVol} />

      {/* SFX/foley markers — each rendered as a Sequence-anchored Audio */}
      {sfx.map((s, i) => {
        const sfxVol = Math.pow(10, s.volumeDb / 20);
        return (
          <Sequence key={i} from={s.startFrame}>
            <Audio src={staticFile(s.file)} volume={sfxVol} />
          </Sequence>
        );
      })}

      {/* Cold open: 0-240 frames (0-8s) covers ColdOpen + Inciting */}
      <Sequence from={0} durationInFrames={HEKAYA2_PHASES.INCITING.end}>
        <ColdOpen
          hero={heroMedia}
          title={title}
          durationFrames={HEKAYA2_PHASES.INCITING.end}
        />
      </Sequence>

      {/* Chapter 1: WANT + OBSTACLE_1 (240-750) */}
      <Sequence
        from={chapters[0].startFrame}
        durationInFrames={chapters[0].durationFrames}
      >
        <Chapter
          cycle={chapters[0]}
          phrases={phrases}
          absoluteStart={chapters[0].startFrame}
        />
      </Sequence>

      {/* Chapter 2: ESCALATION_1 + ESCALATION_2 (750-1350) */}
      <Sequence
        from={chapters[1].startFrame}
        durationInFrames={chapters[1].durationFrames}
      >
        <Chapter
          cycle={chapters[1]}
          phrases={phrases}
          absoluteStart={chapters[1].startFrame}
        />
      </Sequence>

      {/* Silence pivot: 1350-1380 (45-46s) */}
      <Sequence
        from={HEKAYA2_PHASES.SILENCE_PIVOT.start}
        durationInFrames={
          HEKAYA2_PHASES.SILENCE_PIVOT.end - HEKAYA2_PHASES.SILENCE_PIVOT.start
        }
      >
        <SilencePivot pivotImage={chapters[1].photos[chapters[1].photos.length - 1]} />
      </Sequence>

      {/* Chapter 3: CLIMAX + RESOLUTION (1380-1950) */}
      <Sequence
        from={chapters[2].startFrame}
        durationInFrames={chapters[2].durationFrames}
      >
        <Chapter
          cycle={chapters[2]}
          phrases={phrases}
          absoluteStart={chapters[2].startFrame}
        />
      </Sequence>

      {/* Resonance + loop: 1950-2250 (65-75s) */}
      <Sequence
        from={HEKAYA2_PHASES.RESONANCE.start}
        durationInFrames={HEKAYA2_TOTAL_FRAMES - HEKAYA2_PHASES.RESONANCE.start}
      >
        <Resonance
          closingMedia={closingMedia}
          loopHook={loopHook}
          handle={handle}
          sources={sources}
          durationFrames={HEKAYA2_TOTAL_FRAMES - HEKAYA2_PHASES.RESONANCE.start}
        />
      </Sequence>

      {/* Persistent date stamps — same as v1 */}
      <div
        style={{
          position: "absolute",
          bottom: 70,
          left: 50,
          color: HEKAYA_PALETTE.ivory,
          fontFamily: "system-ui, -apple-system, sans-serif",
          fontSize: 16,
          letterSpacing: 4,
          fontWeight: 500,
          opacity: 0.55,
          textTransform: "uppercase",
        }}
      >
        {dateLabel}
      </div>
      <div
        dir="rtl"
        style={{
          position: "absolute",
          bottom: 70,
          right: 50,
          color: HEKAYA_PALETTE.ivory,
          fontFamily: "Tajawal, system-ui, sans-serif",
          fontSize: 18,
          fontWeight: 500,
          opacity: 0.55,
        }}
      >
        {arabicDateLabel}
      </div>
    </AbsoluteFill>
  );
};
```
